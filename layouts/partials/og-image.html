"size" $.size "widths" $.widths)) $.maxWidth }} {{- $words = $words | append $piece }} {{- $piece = "" }} {{- end }} {{- $piece = print $piece . }} {{- end }} {{- $words = $words | append $piece }} {{- end }} {{- range $words }} {{- $w := partial "inline/og-measure.html" (dict "text" .
